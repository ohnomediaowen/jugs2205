  <h1>Water Jug Tracker</h1>

  <div>Full jugs remaining:</div>

  <div class="count" id="count">0</div>


  <button class="use" onclick="useJug()">Jug Changed</button>

  <button class="refill" onclick="refillJug()">Refill Jug</button>

  <button class="reset" onclick="resetJugs()">Reset</button>


  <script>

    const FULL_DEFAULT = 5;


    function load() {

      return Number(localStorage.getItem('jugs') || FULL_DEFAULT);

    }


    function save(n) {

      localStorage.setItem('jugs', n);

    }


    function render() {

      document.getElementById('count').textContent = load();

    }


    function useJug() {

      let n = load();

      if (n > 0) n--;

      save(n);

      render();

    }


    function refillJug() {

      let n = load();

      n++;

      save(n);

      render();

    }


    function resetJugs() {

      save(FULL_DEFAULT);

      render();

    }


    render();


    if ('serviceWorker' in navigator) {

      navigator.serviceWorker.register('sw.js');

    }

  </script>
